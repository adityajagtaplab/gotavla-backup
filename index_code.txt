/* ---------- OFFLINE-SAFE WRITE QUEUE (real disk via AsyncStorage) ---------- */
const PENDING_KEY = 'pendingOps_v1';
async function loadPending() {
  const raw = await AsyncStorage.getItem(PENDING_KEY);
  return raw ? JSON.parse(raw) : [];
}
async function savePending(list) { await AsyncStorage.setItem(PENDING_KEY, JSON.stringify(list)); }
async function enqueue(kind, payload) {
  const list = await loadPending();
  const id = Date.now() + '_' + Math.random().toString(36).slice(2);
  list.push({ id, kind, payload });
  await savePending(list);
  return id;
}
async function dequeue(id) {
  const list = await loadPending();
  await savePending(list.filter((o) => o.id !== id));
}
async function performOp(uid, op) {
  const { kind, payload } = op;
  if (kind === 'addBook') await addDoc(booksCol(uid), payload.data);
  else if (kind === 'updateBook') await updateDoc(doc(booksCol(uid), payload.docId), payload.data);
  else if (kind === 'deleteBook') await deleteDoc(doc(booksCol(uid), payload.docId));
  else if (kind === 'addWishlistKnownId') await setDoc(doc(wishCol(uid), payload.docId), payload.data);
  else if (kind === 'updateWishlist') await updateDoc(doc(wishCol(uid), payload.docId), payload.data);
  else if (kind === 'deleteWishlist') await deleteDoc(doc(wishCol(uid), payload.docId));
  else if (kind === 'addRecycle') await addDoc(recycleCol(uid), payload.data);
  else if (kind === 'deleteRecycle') await deleteDoc(doc(recycleCol(uid), payload.docId));
  else if (kind === 'saveSettings') await setDoc(libDoc(uid), payload.data, { merge: true });
}
async function runOp(uid, kind, payload) {
  const id = await enqueue(kind, payload);
  try {
    await performOp(uid, { kind, payload });
    await waitForPendingWrites(db);
    await dequeue(id);
  } catch (e) {
    // offline — stays queued, replayPending() will retry later
  }
}
async function replayPending(uid) {
  if (!uid) return;
  const list = await loadPending();
  for (const op of list) {
    try {
      await performOp(uid, op);
      await waitForPendingWrites(db);
      await dequeue(op.id);
    } catch (e) { break; } // still offline — stop, try again next time
  }
}

  if (page === 'addbook') {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.pageHead}>
          <TouchableOpacity style={styles.backBtn} onPress={() => setPage(formMode === 'edit' ? 'bookdetail' : 'home')}><Text>←</Text></TouchableOpacity>

        </View>
        <FlatList data={list} keyExtractor={(i) => i.id}

                  else { setCurrentBookId(item.id); setPage('bookdetail'); }

                {isSelecting('books') && (
                  <View style={[styles.selCheck, selected && styles.selCheckOn]}>{selected ? <Text style={{ color: '#fff', fontSize: 12 }}>✓</Text> : null}</View>

                {item.coverPhoto
                  ? <Image source={{ uri: item.coverPhoto }} style={styles.bookCoverImg} />

    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.pageHead}>
          <TouchableOpacity style={styles.backBtn} onPress={() => setPage('home')}><Text>←</Text></TouchableOpacity>
          <Text style={styles.pageTitle}>Settings</Text>
        </View>
        <ScrollView>
          <Text style={styles.label}>ग्रंथालयाचं नाव</Text>
          <TextInput style={styles.fieldInput} defaultValue={settings.libName} onEndEditing={(e) => saveSettingsField('libName', e.nativeEvent.text)} />
          <Text style={styles.label}>About Us</Text>
          <TextInput style={[styles.fieldInput, { height: 70 }]} multiline defaultValue={settings.about} onEndEditing={(e) => saveSettingsField('about', e.nativeEvent.text)} />

          <Text style={[styles.sectionTitle, { marginTop: 16, marginBottom: 8 }]}>Security</Text>
          <TouchableOpacity style={styles.settingsRow} onPress={() => setPage('changepass')}>
            <Text style={styles.bookName}>🔑 Password बदला</Text><Text>›</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.settingsRow} onPress={() => setPage('applock')}>
            <Text style={styles.bookName}>App Lock 🔐</Text>
            <Text style={{ fontSize: 11, color: COLORS.inkSoft }}>{lockLabel}</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.settingsRow} onPress={chooseBackupFolder}>
            <Text style={styles.bookName}>📂 Device Backup Folder</Text>

            <Text style={styles.bookName}>♻️ Recycle Bin</Text><Text>›</Text>

          <TouchableOpacity style={styles.settingsRow} onPress={handleLogout}>

          </TouchableOpacity>
        </ScrollView>
      </SafeAreaView>

  // ================= APP LOCK =================
  if (page === 'applock') {

      ['always', 'बंद करून परत उघडल्यावर लगेच'],

    const setPref = async (v) => { setLockPref(v); await AsyncStorage.setItem('appLockPref', v); };

      <SafeAreaView style={styles.container}>
        <View style={styles.pageHead}>
          <TouchableOpacity style={styles.backBtn} onPress={() => setPage('settings')}><Text>←</Text></TouchableOpacity>
          <Text style={styles.pageTitle}>App Lock</Text>

        <Text style={{ fontSize: 12, color: COLORS.inkSoft, marginBottom: 14 }}>ही निवड फक्त याच device साठी लागू होते.</Text>
        {options.map(([v, label]) => (
          <TouchableOpacity key={v} style={styles.settingsRow} onPress={() => setPref(v)}>
            <Text style={styles.bookName}>{label}</Text>
            <View style={[styles.selCheck, lockPref === v && styles.selCheckOn, { borderColor: COLORS.green }]}>
              {lockPref === v ? <Text style={{ color: '#fff', fontSize: 12 }}>✓</Text> : null}
            </View>
          </TouchableOpacity>
        ))}
      </SafeAreaView>
    );
  }

  // ================= CHANGE PASSWORD =================
  if (page === 'changepass') {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.pageHead}>
          <TouchableOpacity style={styles.backBtn} onPress={() => setPage('settings')}><Text>←</Text></TouchableOpacity>
          <Text style={styles.pageTitle}>Password बदला</Text>
        </View>
        <Text style={styles.label}>सध्याचा Password</Text>
        <TextInput style={styles.fieldInput} secureTextEntry value={oldPass} onChangeText={setOldPass} />
        <Text style={styles.label}>नवीन Password</Text>
        <TextInput style={styles.fieldInput} secureTextEntry value={newPass} onChangeText={setNewPass} />
        <TouchableOpacity style={styles.button} onPress={handleChangePassword}><Text style={styles.buttonText}>Save करा</Text></TouchableOpacity>
      </SafeAreaView>
    );
  }

  // ================= RECYCLE BIN =================
  if (page === 'recyclebin') {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.pageHead}>
          <TouchableOpacity style={styles.backBtn} onPress={() => setPage('settings')}><Text>←</Text></TouchableOpacity>
          <Text style={styles.pageTitle}>Recycle Bin</Text>
        </View>
        <FlatList data={recycle} keyExtractor={(i) => i.id}
          ListEmptyComponent={<Text style={styles.emptyText}>Recycle bin रिकामी आहे</Text>}
          renderItem={({ item }) => (
            <View style={styles.wishRow}>
              <View style={{ flex: 1 }}>
                <Text style={styles.bookName}>{item.type === 'book' ? item.data.name : (item.data.title || 'यादी')}</Text>
                <Text style={styles.bookSub}>{item.type === 'book' ? '📕 पुस्तक' : '📝 Wishlist'}</Text>
              </View>
              <TouchableOpacity onPress={() => restoreItem(item)}><Text style={[styles.linkBtn, { marginRight: 14 }]}>Restore</Text></TouchableOpacity>
              <TouchableOpacity onPress={() => permanentDelete(item)}><Text style={{ color: COLORS.terracotta, fontSize: 12, fontWeight: '700' }}>Delete</Text></TouchableOpacity>
            </View>
          )} />
      </SafeAreaView>
    );
  }

  return null;
}
